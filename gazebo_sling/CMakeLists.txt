cmake_minimum_required(VERSION 3.10)
project(gazebo_sling)
find_package(gazebo REQUIRED)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs)
add_library(sling_cables SHARED src/sling_cables.cpp)
target_compile_features(sling_cables PRIVATE cxx_std_17)
target_include_directories(sling_cables PRIVATE ${GAZEBO_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
target_link_directories(sling_cables PRIVATE ${GAZEBO_LIBRARY_DIRS})
target_link_libraries(sling_cables ${GAZEBO_LIBRARIES} ${catkin_LIBRARIES})
